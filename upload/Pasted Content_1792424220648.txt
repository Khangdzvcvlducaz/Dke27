function migrate(db){
  if(!db)return db;
  if(db.staff&&db.staff.length&&typeof db.staff[0]==='string')db.staff=db.staff.map(function(s){return{name:s,pass:null,role:'staff',view:'pos',lastLogin:null,loginCount:0}});
  if(!db.staff)db.staff=[];
  if(!db.sessions)db.sessions=[];
  if(db.logo===undefined)db.logo=null;
  if(!db.adminName)db.adminName='Admin';
  for(var i=0;i<db.staff.length;i++){var m=db.staff[i];
    if(m.pass===undefined)m.pass=null;
    if(!m.role)m.role='staff';
    if(m.view===undefined)m.view=null;
  }
  if(!db.orders)db.orders=[];
  return db;
}
function findStaffLoose(n){n=(n||'').toLowerCase();
  for(var i=0;i<DB.staff.length;i++)if(DB.staff[i].name.toLowerCase()===n)return DB.staff[i];
  return null}
function findStaffExact(n){for(var i=0;i<DB.staff.length;i++)if(DB.staff[i].name===n)return DB.staff[i];return null}
function touchLogin(name){
  if(!isNameAdmin(name)){
    var m=findStaffLoose(name);
    if(!m){m={name:name,pass:null,role:'staff',view:'pos',lastLogin:null,loginCount:0};DB.staff.push(m)}
    m.lastLogin=Date.now();m.loginCount=(m.loginCount||0)+1;
  }
  logSession(name,'in');
}

var AC=null;
function beep(){try{
  AC=AC||new(window.AudioContext||window.webkitAudioContext)();
  var t=AC.currentTime;
  [0,0.18].forEach(function(off){
    var o=AC.createOscillator(),g=AC.createGain();
    o.type='sine';o.frequency.value=900;
    g.gain.setValueAtTime(0.0001,t+off);
    g.gain.exponentialRampToValueAtTime(0.35,t+off+0.02);
    g.gain.exponentialRampToValueAtTime(0.0001,t+off+0.16);
    o.connect(g);g.connect(AC.destination);o.start(t+off);o.stop(t+off+0.2);
  });
}catch(e){}}

function addToCart(id){
  var m=null;
  for(var i=0;i<DB.menu.length;i++)if(String(DB.menu[i].id)===String(id))m=DB.menu[i];
  if(!m)return;
  var ex=null;
  for(var j=0;j<cart.length;j++)if(String(cart[j].menuId)===String(m.id)&&cart[j].notes.length===0)ex=cart[j];
  if(ex)ex.qty++;else cart.push({menuId:m.id,icon:m.icon,name:m.name,price:m.price,qty:1,notes:[]});

function oCard(o){
  var late=o.status!=='done'&&Date.now()-o.createdAt>8*60*1000;
  var t=new Date(o.createdAt).toLocaleTimeString('vi-VN',{hour:'2-digit',minute:'2-digit'});
  var items='';for(var i=0;i<o.items.length;i++){var it=o.items[i];
    items+='<li><b>'+it.qty+'×</b> '+it.icon+' '+esc(it.name)+(it.notes.length?'<span class="o-notes"> — '+esc(it.notes.join(', '))+'</span>':'')+'</li>'}
  return '<div class="o-card '+o.status+(late?' late':'')+'">'+
    '<div class="o-top"><b class="o-code">'+o.code+'</b>'+
    '<span class="o-type">'+(o.type==='take'?'🥡 Mang đi':'🏠 Tại chỗ')+'</span>'+
    '<span class="o-time">'+t+'</span></div>'+
    '<div class="o-cus">'+esc(o.customer)+(o.by?' · '+esc(o.by):'')+'</div>'+
    '<ul class="o-items">'+items+'</ul>'+
    '<div class="o-total">'+fmt(o.total)+'</div>'+
    '<div class="o-actions">'+
    (o.status==='new'
      ?'<button class="btn btn-primary" data-act="start" data-id="'+o.id+'">Bắt đầu pha</button>'
      :'<button class="btn btn-ok" data-act="finish" data-id="'+o.id+'">✓ Xong món</button>')+
    '<button class="btn btn-danger-ghost" data-act="cancel" data-id="'+o.id+'">✕</button></div></div>';
}

function moneyOf(name,d){
  var paid=DB.orders.filter(function(o){return o.status==='paid'&&o.paidBy===name&&dstr(o.paidAt)===d});
  var sum=0,cash=0,tr=0;
  for(var i=0;i<paid.length;i++){sum+=paid[i].total;if(paid[i].payMethod==='cash')cash+=paid[i].total;else tr+=paid[i].total}
  return{sum:sum,n:paid.length,cash:cash,tr:tr};
}
function renderReport(){
  var d=dstr(),paid=[],canceled=0,active=[],activeSum=0;
  for(var i=0;i<DB.orders.length;i++){var o=DB.orders[i];
    if(o.status==='paid'&&dstr(o.paidAt)===d)paid.push(o);
    else if(o.status==='canceled'&&dstr(o.createdAt)===d)canceled++;
    else if(o.status==='new'||o.status==='making'||o.status==='done'){active.push(o);activeSum+=o.total}}
  var revenue=0,cups=0;
  for(var j=0;j<paid.length;j++){revenue+=paid[j].total;
    for(var k=0;k<paid[j].items.length;k++)cups+=paid[j].items[k].qty}
  $('#stats').innerHTML=

  var cash=0,tr=0,cashN=0,trN=0;
  for(var p=0;p<paid.length;p++){if(paid[p].payMethod==='cash'){cash+=paid[p].total;cashN++}else{tr+=paid[p].total;trN++}}
  $('#paySplit').innerHTML='💵 Tiền mặt: <b>'+cashN+' đơn · '+fmt(cash)+'</b><br>🏦 Chuyển khoản: <b>'+trN+' đơn · '+fmt(tr)+'</b>'+

 $('#logoFile').addEventListener('change',function(){
  var file=this.files&&this.files[0];this.value='';
  if(!file)return;
  var reader=new FileReader();
  reader.onload=function(){
    var img=new Image();
    img.onload=function(){
      try{
        var MAX=560,sc=Math.min(1,MAX/Math.max(img.width,img.height));
        var cv=document.createElement('canvas');
        cv.width=Math.round(img.width*sc);cv.height=Math.round(img.height*sc);
        cv.getContext('2d').drawImage(img,0,0,cv.width,cv.height);
        var data=cv.toDataURL('image/png');
        if(data.length>2500000)return toast('Ảnh quá lớn, hãy chọn ảnh nhỏ hơn',true);
        DB.logo=data;save(true);paintBrand();toast('✓ Logo đã lưu lên cloud — mọi thiết bị sẽ thấy');
      }catch(e){toast('Không xử lý được ảnh này')}
    };
    img.onerror=function(){toast('File không phải ảnh hợp lệ')};
    img.src=reader.result;
  };
  reader.readAsDataURL(file);
});
